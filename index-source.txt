    function choose(key, value) {
      state[key] = value;
      hasCopiedQuote = false;
      if (key === 'model') {
        if (value) activeGuideModel = value;
        state.size = null; state.finish = null; state.headConfig = null;
        state.table = null; state.moodLights = null; state.mattress = null; state.delivery = null;
        if (value === 'shuve') state.finish = '보에르 - Fog(아이보리)';
      }
      if (key === 'size' || key === 'finish') {
        state.headConfig = null; state.table = null;
        state.moodLights = null; state.mattress = null; state.delivery = null;
      }
      if (key === 'size' && value === 'SS+SS 커플') state.table = null;
      saveState(); render();
    }

    function frameItem() {
      if (!state.model || !state.size || !state.finish) return null;
      return bySection('frames').find(item => modelMatches(item) && item.size === state.size && (state.model === 'shuve' || item.description.includes(state.finish))) || null;
    }
    function isFrameComplete() { return Boolean(state.model && state.size && state.finish && frameItem()); }
    function fabricItem() {
      if (state.model !== 'shuve' || state.finish === '보에르 - Fog(아이보리)') return null;
      if (state.finish?.startsWith('스웨린 Suerin - ')) {
        return bySection('shuveFabrics').find(item => item.description.includes('스웨린 Suerin')) || null;
      }
      return bySection('shuveFabrics').find(item => item.description === state.finish) || null;
    }
    function shuveCustomFinishChoices() {
      return bySection('shuveFabrics').flatMap(item => {
        if (item.description.includes('스웨린 Suerin')) {
          return SUERIN_VARIANTS.map(variant => ({
            ...variant,
            swatch: fabricSwatch(variant.value),
            note: item.salePrice ? plusMoney(item.salePrice) : '추가 없음',
          }));
        }
        return [{
          value: item.description,
          label: item.description,
          swatch: fabricSwatch(item.description),
          note: item.salePrice ? plusMoney(item.salePrice) : '추가 없음',
        }];
      });
    }
    function mattressKind(item) {
      if (item.description.includes('하루끝')) return 'MH';
      if (item.description.includes('미디움')) return 'M';
      return 'H';
    }
    function mattressItems() {
      const token = tokenSize(state.size);
      const pattern = new RegExp(`(^|: )${escapeRegex(token)}\\s*\\/`);
      const map = new Map();
      bySection('mattresses').filter(item => normalizedItemSize(item) === token || pattern.test(item.description)).forEach(item => {
        const kind = mattressKind(item);
        if (!map.has(kind)) map.set(kind, item);
      });
      return map;
    }
    function headConfigItem(config = state.headConfig) {
      if (config === 'none') return null;
      const section = state.model === 'mabed' ? 'mabedHeads' : 'shuveHeads';
      const token = tokenSize(state.size);
      const candidates = bySection(section).filter(item => normalizedItemSize(item) === token && (state.model === 'shuve' || item.description.includes(`/ ${state.finish}`)));
      if (config === 'head') return candidates.find(item => !item.description.includes('+ 협탁')) || null;
      if (config === 'one') return candidates.find(item => item.description.includes('+ 협탁1')) || null;
      if (config === 'two') return candidates.find(item => item.description.includes('+ 협탁2')) || null;
      return null;
    }
    function isAutomaticTable(item) { return item.description.startsWith('자동 테이블') || /\/ 자동\s*$/.test(item.description); }
    function minimumSalePrice(sections, predicate = () => true) {
      const names = Array.isArray(sections) ? sections : [sections];
      const values = names.flatMap(section => bySection(section)).filter(predicate).map(item => Number(item.salePrice)).filter(Number.isFinite);
      return values.length ? Math.min(...values) : null;
    }

    }

    function calculatedDimensions() {
      if (!isFrameComplete()) return null;
      const guide = SIZE_GUIDE[state.model];
      const baseWidth = guide.widths[state.size] || 0;
      const nightstandCount = state.headConfig === 'one' ? 1 : state.headConfig === 'two' ? 2 : 0;
      const nightstandExtra = nightstandCount * guide.nightstandWidth;
      const tableExtra = ['manual', 'auto'].includes(state.table) ? guide.tableWidth : 0;
      // 협탁과 테이블은 침대의 서로 다른 위치에 놓이므로, 실제 설치 폭은 두 추가 폭을 더하지 않고 더 큰 쪽을 적용합니다.
      const widthExtra = Math.max(nightstandExtra, tableExtra);
      const hasHead = ['head', 'one', 'two'].includes(state.headConfig);
      const headExtra = hasHead ? guide.headDepth : 0;
      const autoTableExtra = state.table === 'auto' ? guide.autoTableDepth : 0;
      const width = baseWidth + widthExtra;
      const depth = guide.frameDepth + headExtra + autoTableExtra;
      const mattress = MATTRESS_DIMENSIONS[state.size] || null;
      const widthParts = [`프레임 ${baseWidth}cm`];
      if (nightstandExtra) widthParts.push(`협탁 ${nightstandCount}개 ${nightstandExtra}cm`);
      if (tableExtra) widthParts.push(`테이블 여유 ${tableExtra}cm`);
      const depthParts = [`프레임 ${guide.frameDepth}cm`];
      if (headExtra) depthParts.push(`헤드 ${headExtra}cm`);
      if (autoTableExtra) depthParts.push(`자동 레일 ${autoTableExtra}cm`);
      return {
        guide, baseWidth, width, depth, mattress, nightstandCount,
        widthParts, depthParts,
        manualNote: state.table === 'manual' ? '수동 테이블을 움직여 사용하는 여유 공간은 설치 장소에서 별도로 확인해 주세요.' : '',
        widthRuleNote: nightstandExtra && tableExtra ? '협탁과 테이블은 서로 다른 위치에 있어 더 넓은 쪽을 기준으로 전체 가로 폭을 계산했습니다.' : '',
      };
    }

    function quoteLines() {
      const lines = [];
      if (!isFrameComplete()) return lines;
      const frame = frameItem();
      if (frame) lines.push({ label: `${MODEL_LABEL[state.model]} ${state.size} 프레임`, detail: state.finish, price: frame.salePrice });
      const fabric = fabricItem();
      if (fabric) lines.push({ label: '커스텀 패브릭', detail: state.finish, price: fabric.salePrice });
      const mattress = state.mattress === 'none' ? null : mattressItems().get(state.mattress);
      if (mattress) {
        const label = state.mattress === 'MH' ? '하루끝 스프링 매트리스' : state.mattress === 'H' ? '라텍스폼 하드' : '라텍스폼 미디움';
        lines.push({ label, detail: tokenSize(state.size), price: mattress.salePrice });
      }
      const head = headConfigItem();
      if (head) {
        const label = state.headConfig === 'head' ? '헤드' : state.headConfig === 'one' ? '헤드 + 협탁 한쪽' : '헤드 + 협탁 양쪽';
        const direction = state.headConfig === 'one' ? ' · 설치 방향 상담 시 결정' : '';
        lines.push({ label, detail: `${state.finish}${direction}`, price: head.salePrice });
      }
      const table = tableItem();
      if (table) lines.push({ label: state.table === 'auto' ? '자동 테이블' : '수동 테이블', detail: state.table === 'auto' ? '전동 레일 포함' : '손으로 위치 조절', price: table.salePrice });
      const mood = moodItem();
      if (mood) {
        const detail = state.moodLights === 1 ? '프레임 하부 좌우 중 한 곳 · 설치 시 기사님과 위치 협의' : '프레임 하부 좌우에 모두 설치';
        lines.push({ label: state.moodLights === 1 ? '인체감지 무드등 한쪽' : '인체감지 무드등 양쪽', detail, price: mood.salePrice });
      }
      const delivery = deliveryItem();
      if (delivery) lines.push({ label: '배송 추가비', detail: delivery.description, price: delivery.salePrice });
      return lines;
    }

    function optionCard(choice, current, onSelect) {
      const card = document.createElement('div');
      card.className = `option-card${choice.value === current ? ' selected' : ''}${choice.disabled ? ' unavailable' : ''}`;
      const wholeClickable = !choice.disabled && choice.clickWholeCard !== false;
      if (wholeClickable) card.classList.add('whole-clickable');
      if (choice.image) {
        const image = document.createElement('img');
        image.className = 'product-image'; image.src = choice.image; image.alt = choice.imageAlt || '';
        card.appendChild(image);
      }
      const select = document.createElement('button');
      select.type = 'button'; select.className = 'option-select';
      select.setAttribute('aria-pressed', choice.value === current ? 'true' : 'false');
      select.disabled = Boolean(choice.disabled);
      if (choice.disabled) select.setAttribute('aria-disabled', 'true');
      const priceLabel = choice.priceText ?? plusMoney(choice.price);
      const selectAction = choice.frameGate ? '프레임 사양 먼저 선택' : choice.disabled ? '선택 불가' : choice.value === current ? '✓ 선택됨' : '선택하기';
      select.innerHTML = `${choice.tag ? `<span class="option-tag">${choice.tag}</span>` : ''}<span class="option-name">${choice.label}</span>${choice.type ? `<span class="product-type">${choice.type}</span>` : ''}${choice.summary ? `<span class="option-summary">${choice.summary}</span>` : ''}${priceLabel ? `<span class="option-price">${priceLabel}</span>` : ''}${choice.showSelectButton ? `<span class="option-card-action">${selectAction}</span>` : ''}`;
      const activate = () => {
        if (choice.frameGate) { $('step-frame').scrollIntoView({ behavior: 'smooth', block: 'start' }); return; }
        onSelect(choice.value);
      };
      select.addEventListener('click', activate);
      card.appendChild(select);
      if (choice.detailKey) {
        const detail = DETAILS[choice.detailKey];
        const explainer = document.createElement('div');
        explainer.className = 'option-explainer';
        explainer.innerHTML = `${choice.inlineVisual === false ? '' : componentVisual(choice.detailKey, true)}${detail ? detailBody(detail) : ''}`;
        card.appendChild(explainer);
      }
      if (wholeClickable) {
        card.addEventListener('click', event => {
          if (event.target.closest('.option-select')) return;
          activate();
        });
      }
      return card;
    }
    function renderOptionGrid(id, choices, current, onSelect, className = '') {
      const root = $(id); root.innerHTML = ''; root.className = `option-grid ${className}`.trim();
      choices.forEach(choice => root.appendChild(optionCard(choice, current, onSelect)));
      setupOptionCarousel(root);
    }

    function setupOptionCarousel(root) {
      if (root.dataset.carousel !== 'true') return;
      let controls = root.nextElementSibling;
      if (!controls?.classList.contains('carousel-controls')) {
        controls = document.createElement('div');
        controls.className = 'carousel-controls';
        controls.innerHTML = `<button class="carousel-arrow prev" type="button" aria-label="이전 선택지">‹</button><span class="carousel-status" aria-live="polite"></span><button class="carousel-arrow next" type="button" aria-label="다음 선택지">›</button>`;
        root.insertAdjacentElement('afterend', controls);
      }
      const cards = () => [...root.querySelectorAll('.option-card')];
      const cardLeft = card => card.getBoundingClientRect().left - root.getBoundingClientRect().left + root.scrollLeft;
      const currentIndex = () => {
        const list = cards();
        if (!list.length) return 0;
        return list.reduce((best, card, index) => Math.abs(cardLeft(card) - root.scrollLeft) < best.distance
          ? { index, distance: Math.abs(cardLeft(card) - root.scrollLeft) }
          : best, { index: 0, distance: Infinity }).index;
      };
      const update = () => {
        const list = cards();
        controls.hidden = list.length < 2;
        controls.querySelector('.carousel-status').textContent = list.length ? `${currentIndex() + 1} / ${list.length} · 옆으로 넘겨보세요` : '';
      };
      const move = delta => {
        const list = cards();
        if (!list.length) return;
        const targetIndex = (currentIndex() + delta + list.length) % list.length;
        root.scrollTo({ left: cardLeft(list[targetIndex]), behavior: 'smooth' });
        window.setTimeout(update, 360);
      };
      if (root.dataset.carouselBound !== 'true') {
        root.dataset.carouselBound = 'true';
        controls.querySelector('.prev').addEventListener('click', () => move(-1));
        controls.querySelector('.next').addEventListener('click', () => move(1));
        let startX = null;
        let startIndex = 0;
        root.addEventListener('touchstart', event => {
          startX = event.touches[0]?.clientX ?? null;
          startIndex = currentIndex();
        }, { passive: true });
        root.addEventListener('touchend', event => {
          if (startX == null) return;
          const endX = event.changedTouches[0]?.clientX ?? startX;
          const distance = endX - startX;
          const list = cards();
          if (Math.abs(distance) > 42 && list.length > 1) {
            if (startIndex === list.length - 1 && distance < 0) root.scrollTo({ left: cardLeft(list[0]), behavior: 'smooth' });
            if (startIndex === 0 && distance > 0) root.scrollTo({ left: cardLeft(list[list.length - 1]), behavior: 'smooth' });
          }
          startX = null;
          window.setTimeout(update, 360);
        }, { passive: true });
        let scrollTimer = null;
        root.addEventListener('scroll', () => {
          clearTimeout(scrollTimer);
          scrollTimer = window.setTimeout(update, 90);
        }, { passive: true });
      }
      requestAnimationFrame(update);
    }

      const button = $('imageSaveButton');

      button.disabled = true;
      button.textContent = '이미지 만드는 중…';
      document.body.classList.add('saving-quote-image');
      try {
        const fieldIds = ['customerName', 'customerPhone', 'customerPostcode', 'customerAddress', 'customerAddressDetail', 'deliveryRequest'];
        const mobileOrInApp = /Android|iPhone|iPad|iPod|KAKAOTALK/i.test(navigator.userAgent);
        const canvas = await window.html2canvas($('receipt'), {
          backgroundColor: '#ffffff',
          scale: mobileOrInApp ? 1.25 : 2,
          useCORS: true,
          logging: false,
          onclone(clonedDocument) {
            fieldIds.forEach(id => {
              const original = $(id);
              const cloned = clonedDocument.getElementById(id);
              if (!original || !cloned) return;
              cloned.value = original.value;
              if (cloned.tagName === 'TEXTAREA') cloned.textContent = original.value;
              else cloned.setAttribute('value', original.value);
            });
          },
        });
        const date = new Date().toISOString().slice(0, 10);
        const filename = `버킷츠-${MODEL_LABEL[state.model]}-${state.size}-견적-${date}.png`;
        if (mobileOrInApp) {
          const dataUrl = canvas.toDataURL('image/png');
          openSheet({
            title: '견적 이미지가 완성됐어요',
            body: '<p class="image-save-help">아래 <strong>PNG 이미지 다운로드</strong> 버튼을 누르세요. 저장된 파일은 휴대폰의 알림창이나 다운로드 폴더에서 확인할 수 있습니다.</p><a class="image-download-link" id="quoteImageDownloadLink" href="#" download>PNG 이미지 다운로드</a><img class="quote-image-preview" id="quoteImagePreview" alt="완성된 버킷츠 침대 견적 이미지">',
            hideAction: true,
          });
          const downloadLink = $('quoteImageDownloadLink');
          downloadLink.href = dataUrl;
          downloadLink.download = filename;
          downloadLink.addEventListener('click', () => showToast('다운로드를 요청했습니다. 알림창이나 다운로드 폴더를 확인해 주세요.'));
          $('quoteImagePreview').src = dataUrl;
          showToast('PNG 이미지 다운로드 버튼을 눌러 저장하세요.');
        } else {
          const blob = await new Promise(resolve => canvas.toBlob(resolve, 'image/png'));
          if (!blob) throw new Error('PNG 생성 실패');
          const url = URL.createObjectURL(blob);
          const link = document.createElement('a');
          link.href = url;
          link.download = filename;
          document.body.appendChild(link);
          link.click();
          link.remove();
          setTimeout(() => URL.revokeObjectURL(url), 3000);
          showToast('견적 이미지를 저장했습니다.');
        }
      } catch (_) {
        showToast('이미지 저장에 실패했습니다. 잠시 후 다시 시도해 주세요.');
      } finally {
        document.body.classList.remove('saving-quote-image');
        button.textContent = originalLabel;
        button.disabled = !isFrameComplete();
      }
    }

    function updateProgress(step) {

    function setupProgressObserver() {
      if (!('IntersectionObserver' in window)) return;
      const observer = new IntersectionObserver(entries => {
        const visible = entries.filter(entry => entry.isIntersecting).sort((a,b) => b.intersectionRatio - a.intersectionRatio)[0];
        if (visible) updateProgress(visible.target.dataset.step);
      }, { rootMargin: '-28% 0px -58% 0px', threshold: [0, .2, .5] });
      document.querySelectorAll('[data-step]').forEach(section => observer.observe(section));
    }

    function loadLiveCatalog() {
      const url = String(CONFIG.appsScriptUrl || '').trim();
      if (!/^https:\/\/script\.google\.com\/macros\/s\/.+\/exec(?:\?.*)?$/.test(url)) return;
      const callback = `__motionBedCatalog_${Date.now()}`;
      const script = document.createElement('script'); let settled = false;
      const finish = () => { if (settled) return; settled = true; delete window[callback]; script.remove(); };
      const timer = setTimeout(finish, CONFIG.jsonpTimeoutMs);
      window[callback] = payload => { clearTimeout(timer); if (payload?.ok && Array.isArray(payload.items)) { catalog = payload; hasCopiedQuote = false; render(); } finish(); };
      script.onerror = () => { clearTimeout(timer); finish(); };
      script.src = `${url}${url.includes('?') ? '&' : '?'}callback=${encodeURIComponent(callback)}&_=${Date.now()}`;
      document.head.appendChild(script);
    }
